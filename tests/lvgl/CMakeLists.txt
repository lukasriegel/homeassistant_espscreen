cmake_minimum_required(VERSION 3.16)
project(overlay_scroll_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT LVGL_SOURCE_DIR)
  message(FATAL_ERROR "Pass -DLVGL_SOURCE_DIR=<LVGL 9 source directory>")
endif()
add_compile_definitions(LV_CONF_SKIP LV_MEM_SIZE=131072)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
add_executable(test_overlay_scroll test_overlay_scroll.cpp)
target_link_libraries(test_overlay_scroll PRIVATE lvgl::lvgl)
enable_testing()
add_test(NAME overlay_scroll COMMAND test_overlay_scroll)
